rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Реакции: читать могут все, писать — только авторизованные
    match /reactions/{lessonId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Голоса пользователей (чтобы отображать активную кнопку)
    match /votes/{voteId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == request.resource.data.userId;
    }
    
    // Черновики: читать могут все, писать — только модераторы
    match /config/drafts {
      allow read: if true;
      // Замените email на свой!
      allow write: if request.auth != null && 
                     request.auth.token.email in ['31arseni@mail.ru', 'admin@example.com'];
    }
  }
}
